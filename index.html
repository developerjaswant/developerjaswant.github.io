namespace App\Services\Payment;

class PaymentGatewayService
{
    public function processPayment(GatewayContract $gateway, array $payload): PaymentResponse
    {
        $charge = $gateway->charge($payload);

        return $this->syncTransactionStatus(
            $charge->status,
            $charge->transaction_id,
            $charge->meta
        );
    }
}
